cmake_minimum_required(VERSION 3.24)
project(HRLH VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
option(HRLH_WITH_GUROBI "Enable formal stack preprocessing and complete exact model" ON)
set(GUROBI_HOME "$ENV{GUROBI_HOME}" CACHE PATH "Gurobi SDK installation")
if(MSVC)
  add_compile_options(/utf-8 /EHsc)
endif()
if(HRLH_WITH_GUROBI)
  find_path(GUROBI_INCLUDE gurobi_c++.h HINTS "${GUROBI_HOME}/include" REQUIRED)
  find_library(GUROBI_C NAMES gurobi130 gurobi120 gurobi110 HINTS "${GUROBI_HOME}/lib" REQUIRED)
  if(NOT GUROBI_CPP)
    find_library(GUROBI_CPP_RELEASE NAMES gurobi_c++md2017 gurobi_c++md2019 gurobi_c++md2022 gurobi_c++ HINTS "${GUROBI_HOME}/lib" REQUIRED)
    if(MSVC)
      find_library(GUROBI_CPP_DEBUG NAMES gurobi_c++mdd2017 gurobi_c++mdd2019 gurobi_c++mdd2022 HINTS "${GUROBI_HOME}/lib" REQUIRED)
      set(GUROBI_CPP optimized "${GUROBI_CPP_RELEASE}" debug "${GUROBI_CPP_DEBUG}")
    else()
      set(GUROBI_CPP "${GUROBI_CPP_RELEASE}")
    endif()
  endif()
endif()
set(ILS_SEARCH Activity CostEvaluator DynamicBitset Client ClientGroup Depot Location VehicleType ProblemData RandomNumberGenerator Route Solution LoadSegment DurationSegment search/LocalSearch search/neighbourhood search/PerturbationManager search/Route search/SearchSpace search/Solution search/SwapTails case_protocol solver_port)
list(TRANSFORM ILS_SEARCH PREPEND "src/ils_search/")
list(TRANSFORM ILS_SEARCH APPEND ".cpp")
add_library(hrlh_ils_search STATIC ${ILS_SEARCH})
target_include_directories(hrlh_ils_search PUBLIC include/ils_search)
add_library(hrlh_model STATIC src/common/problem_model.cpp)
target_include_directories(hrlh_model PUBLIC include/common)
add_library(hrlh_stacks STATIC src/common/stack_preprocessing.cpp)
target_include_directories(hrlh_stacks PUBLIC include/common)
if(HRLH_WITH_GUROBI)
  target_compile_definitions(hrlh_model PUBLIC HRLH_MODEL_WITH_GUROBI)
  target_include_directories(hrlh_model PUBLIC "${GUROBI_INCLUDE}")
  target_link_libraries(hrlh_model PUBLIC ${GUROBI_CPP} "${GUROBI_C}")
  target_sources(hrlh_model PRIVATE src/exact/exact_model.cpp)
  target_compile_definitions(hrlh_stacks PUBLIC HRLH_STACKS_WITH_GUROBI)
  target_include_directories(hrlh_stacks PUBLIC "${GUROBI_INCLUDE}")
  target_link_libraries(hrlh_stacks PUBLIC ${GUROBI_CPP} "${GUROBI_C}")
endif()
add_library(hrlh_ils_engine STATIC src/ils/hrlh.cpp src/common/optional_demand_profit_intensification.cpp src/common/loading_bounds.cpp src/framework/independent_checker.cpp)
target_include_directories(hrlh_ils_engine PUBLIC include/ils include/framework include/common)
target_compile_definitions(hrlh_ils_engine PUBLIC HRLH_WITH_ILS_SEARCH)
target_link_libraries(hrlh_ils_engine PUBLIC hrlh_model hrlh_stacks hrlh_ils_search)
add_library(hrlh_framework STATIC src/framework/hrlh.cpp src/common/optional_demand_profit_intensification.cpp src/common/loading_bounds.cpp src/framework/routing_problem.cpp src/framework/alns_destroy.cpp src/framework/alns_repair.cpp src/framework/alns.cpp src/framework/independent_checker.cpp)
target_include_directories(hrlh_framework PUBLIC include/framework include/common)
target_compile_definitions(hrlh_framework PUBLIC HRLH_WITH_ILS_SEARCH)
target_link_libraries(hrlh_framework PUBLIC hrlh_model hrlh_stacks hrlh_ils_search)
add_library(hrlh_hgs_search STATIC src/hgs/Params.cpp src/hgs/Individual.cpp src/hgs/Split.cpp src/hgs/PackingAwareMoves.cpp src/hgs/LocalSearch.cpp src/hgs/Population.cpp src/hgs/Genetic.cpp src/framework/hgs.cpp)
target_include_directories(hrlh_hgs_search PUBLIC include/hgs)
target_link_libraries(hrlh_hgs_search PUBLIC hrlh_framework)
add_executable(hrlh_ils src/main.cpp)
target_link_libraries(hrlh_ils PRIVATE hrlh_ils_engine)
foreach(backend IN ITEMS alns hgs)
  add_executable(hrlh_${backend} src/main.cpp)
  string(TOUPPER ${backend} upper_backend)
  target_compile_definitions(hrlh_${backend} PRIVATE HRLH_ROUTING_COMPARISON HRLH_BACKEND="${upper_backend}")
  if(backend STREQUAL "alns")
    target_compile_definitions(hrlh_${backend} PRIVATE HRLH_BACKEND_ALNS)
    target_link_libraries(hrlh_${backend} PRIVATE hrlh_framework)
  else()
    target_link_libraries(hrlh_${backend} PRIVATE hrlh_hgs_search)
  endif()
endforeach()
if(HRLH_WITH_GUROBI)
  add_executable(hrlh_exact src/exact/main.cpp)
  target_include_directories(hrlh_exact PRIVATE include/framework)
  target_link_libraries(hrlh_exact PRIVATE hrlh_model)
endif()
